# ------------------------------------------------------------------------
# Pile ML : deux jeux de contraintes mutuellement exclusifs.
#
# macOS Intel (x86_64) : PyTorch ne publie plus de wheel x86_64 au-delà de
# 2.2.2, ce qui plafonne toute la pile. Sans ces bornes, pip installe
# transformers 5.x (qui exige torch >= 2.5), désactive torch en silence, puis
# échoue sur « NameError: name 'nn' is not defined ». numpy doit rester en 1.x :
# torch 2.2.2 est compilé contre l'ABI numpy 1.
#
# ⚠️ Mac Intel : PYTHON 3.12 MAXIMUM. Les wheels macOS x86_64 de torch 2.2.2
# (cp38-cp312) et de numpy 1.26.4 (cp39-cp312) ne couvrent pas Python 3.13+.
# Sous 3.13, pip ne trouve rien d'installable et tente une compilation qui
# echoue. launch.sh verifie ce point et s'arrete avec un message explicite.
#
# Partout ailleurs (Windows, Linux, Apple Silicon) : bornes hautes majeures
# calées sur la pile validée en local (torch 2.8, transformers 4.56,
# sentence-transformers 5.1, numpy 2.3) pour qu'une installation neuve ne
# ramasse pas « la dernière version de tout ».
# ------------------------------------------------------------------------

# --- macOS Intel (x86_64) ---
torch<=2.2.2; platform_system == "Darwin" and platform_machine == "x86_64"
numpy>=1.24.0,<2; platform_system == "Darwin" and platform_machine == "x86_64"
transformers>=4.40,<5; platform_system == "Darwin" and platform_machine == "x86_64"
sentence-transformers>=2.2.0,<4; platform_system == "Darwin" and platform_machine == "x86_64"
peft>=0.4.0,<0.14; platform_system == "Darwin" and platform_machine == "x86_64"

# --- Windows / Linux / Apple Silicon ---
torch>=2.0.0,<3; platform_system != "Darwin" or platform_machine != "x86_64"
numpy>=1.24.0,<3; platform_system != "Darwin" or platform_machine != "x86_64"
# transformers est tiré par sentence-transformers ; on l'épingle explicitement
# car c'est lui qui pilote la contrainte torch de toute la pile.
transformers>=4.40,<5; platform_system != "Darwin" or platform_machine != "x86_64"
